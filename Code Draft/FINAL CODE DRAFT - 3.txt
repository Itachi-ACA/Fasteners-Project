# =================================================================================
# ROUNDING FUNCTION
# Odd lengths → round UP to nearest even 10
# e.g.  X25 → X30,  X35 → X40,  X45 → X50,  X55 → X60,  X65 → X70
# =================================================================================

def round_length(length_str):
    """Round odd bolt lengths up to nearest 10."""
    try:
        length = int(length_str)
        return math.ceil(length / 10) * 10
    except:
        return None


# =================================================================================
# WEIGHT LOOKUP FUNCTION
# NO fake proxy for MAGNI. If coating not in table → N/A
# =================================================================================

def get_weight_info(size_material: str, qty: int):
    """
    Looks up pcs/kg for a given fastener key and calculates total weight.
    Returns (pcs_per_kg, total_weight_kg) or ("N/A", "N/A") if not found.
    MAGNI, BLACK, SS, HT will return N/A unless explicitly added to WEIGHT_TABLE.
    """
    key = size_material.upper().strip()

    # Strip stud prefix  e.g. "S-M16X50-HDG" → "M16X50-HDG"
    key = re.sub(r"^S-", "", key)

    # Apply rounding to the length portion
    # e.g. M12X55-HDG → M12X60-HDG
    m = re.match(r"(M\d+)X(\d+)(-[A-Z]+)", key)
    if m:
        size    = m.group(1)          # e.g. M12
        length  = m.group(2)          # e.g. 55
        coating = m.group(3)          # e.g. -HDG
        rounded = round_length(length)
        if rounded is None:
            return ("N/A", "N/A")
        key = f"{size}X{rounded}{coating}"

    pcs_per_kg = WEIGHT_TABLE.get(key, None)

    if pcs_per_kg is None:
        return ("N/A", "N/A")

    total_weight = round(qty / pcs_per_kg, 2)
    return (pcs_per_kg, total_weight)


# =================================================================================
# ROUNDING HELPER FOR DISPLAY IN OUTPUT TABLE
# The displayed size in bolt/stud table should also show the ROUNDED length
# e.g. input M12X55-HDG → displayed as M12X60-HDG in output
# =================================================================================

def round_size_material(size_material: str) -> str:
    """Return the size_material with length rounded to nearest 10.
    Also converts coating to SS for sizes below M10 (M4, M5, M6, M8)."""
    key = size_material.upper().strip()
    prefix = ""
    if key.startswith("S-"):
        prefix = "S-"
        key = key[2:]
    m = re.match(r"(M\d+)X(\d+)(-[A-Z]+)", key)
    if m:
        size    = m.group(1)
        length  = m.group(2)
        coating = m.group(3)
        size_num = int(re.search(r"\d+", size).group())
        if size_num < 10:
            coating = "-SS"
        rounded = round_length(length)
        if rounded:
            return f"{prefix}{size}X{rounded}{coating}"
    return size_material


def is_below_m10(size_material: str) -> bool:
    """Returns True if fastener diameter is below M10 (M4, M5, M6, M8)."""
    key = re.sub(r"^S-", "", size_material.upper().strip())
    m = re.search(r"M(\d+)", key)
    if m:
        return int(m.group(1)) < 10
    return False


# ================= PDF READ =================
def extract_pdf_lines(file_path):
    lines = []
    with pdfplumber.open(file_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines.extend(text.split("\n"))
    return lines


# ================= HEADER =================
def extract_header_info(lines):
    header = {"WORK_ORDER": "", "REVISION": "", "CUSTOMER": "", "RATING": ""}
    for line in lines:
        u = line.upper()
        if "WORK ORDER" in u:
            m = re.search(r"WORK ORDER\s+(\d+)", u)
            if m:
                header["WORK_ORDER"] = m.group(1)
        if "REVISION" in u:
            m = re.search(r"REVISION\s+([A-Z0-9]+)", u)
            if m:
                header["REVISION"] = m.group(1)
        if "CUSTOMER" in u:
            # PDF extracts as "CUSTOMERM/S. ABN..." (no space) or "CUSTOMER M/S. ABN..."
            m = re.search(r"CUSTOMER\s*(.*)", line, re.IGNORECASE)
            if m:
                name = m.group(1).strip()
                if name:
                    header["CUSTOMER"] = name
        if "RATING" in u:
            match = re.search(r"RATING\s*(.*?MVA.*?kV)", line, re.IGNORECASE)
            if match:
                header["RATING"] = match.group(1).strip()
    return header


# ================= HARDWARE PARSER =================
def parse_hardware(lines):
    data = []
    for line in lines:
        if re.search(r"(BOLT|STUD|SCREW)", line.upper()) and re.search(r"M\d+X\d+", line.upper()):
            parts = line.split()
            for i, part in enumerate(parts):
                if re.match(r"S?-?M\d+X\d+-?[A-Z]*", part.upper()):
                    try:
                        # Treat SCREW same as BOLT
                        fastener = parts[i - 1].upper()
                        if fastener == "SCREW":
                            fastener = "BOLT"
                        data.append({
                            "FASTENER"     : fastener,
                            "SIZE_MATERIAL": part.upper(),
                            "QTY"          : int(parts[i + 1]),
                            "NUT"          : int(parts[i + 2]),
                            "L_NUT"        : int(parts[i + 3]),
                            "P_WASHER"     : int(parts[i + 4]),
                            "S_WASHER"     : int(parts[i + 5])
                        })
                    except:
                        continue
    return pd.DataFrame(data)


# ================= SUMMARY =================
def build_summary(df):
    summary = defaultdict(lambda: {
        "FULL_NUT": 0, "LOCK_NUT": 0,
        "PLAIN_WASHER": 0, "SPRING_WASHER": 0
    })
    unknown_materials = set()

    for _, row in df.iterrows():
        size_match = re.search(r"M\d+", row["SIZE_MATERIAL"].upper())
        if not size_match:
            continue
        size     = size_match.group(0)
        material = "HDG"

        material_match = re.search(r"-([A-Z]+)$", row["SIZE_MATERIAL"].upper())
        if material_match:
            extracted = material_match.group(1)
            if extracted in ["HDG", "MAGNI"]:
                material = extracted
            else:
                unknown_materials.add(extracted)
        else:
            unknown_materials.add("UNKNOWN")

        key = (size, material)
        summary[key]["FULL_NUT"]     += row["NUT"]
        summary[key]["LOCK_NUT"]     += row["L_NUT"]
        summary[key]["PLAIN_WASHER"] += row["P_WASHER"]
        summary[key]["SPRING_WASHER"]+= row["S_WASHER"]

    return summary, unknown_materials

# ================= SELECT FILE =================
def select_file():
    file_path = filedialog.askopenfilename(
        title="Select PDF",
        filetypes=[("PDF Files", "*.pdf")]
    )
    if not file_path:
        return

    lines  = extract_pdf_lines(file_path)
    header = extract_header_info(lines)
    df     = parse_hardware(lines)

    if df.empty:
        messagebox.showerror("Error", "No hardware data found.")
        return

    summary, unknown_materials = build_summary(df)

    if unknown_materials:
        messagebox.showwarning(
            "Material Warning",
            f"Unknown material detected:\n{', '.join(unknown_materials)}"
        )

    export_excel(header, df, summary, open_var.get())
    messagebox.showinfo("Success", "Hardware Output Generated Successfully!")

         font=("Segoe UI", 24, "bold"), fg=ACCENT, bg=LIGHT_BG).pack(pady=(80, 20))

progress = ttk.Progressbar(main_panel, mode="indeterminate", length=300)

def start_loading():
    progress.pack(pady=10)
    progress.start(10)

def stop_loading():
    progress.stop()
    progress.pack_forget()

def process_with_loader():
    start_loading()
    root.after(200, run_backend)

def run_backend():
    try:
        select_file()
    finally:
        stop_loading()

                       relief="flat", width=26, height=2, cursor="hand2")
select_btn.pack()

select_btn.bind("<Leave>", hover_out)

# ================= PDF READ =================
def extract_pdf_lines(file_path):
    lines = []
    with pdfplumber.open(file_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines.extend(text.split("\n"))
    return lines

# ================= HEADER =================
def extract_header_info(lines):
    header = {"WORK_ORDER": "", "REVISION": "", "CUSTOMER": "", "RATING": ""}
    for line in lines:
        u = line.upper()
        if "WORK ORDER" in u:
            m = re.search(r"WORK ORDER\s+(\d+)", u)
            if m:
                header["WORK_ORDER"] = m.group(1)
        if "REVISION" in u:
            m = re.search(r"REVISION\s+([A-Z0-9]+)", u)
            if m:
                header["REVISION"] = m.group(1)
        if "CUSTOMER" in u:
            m = re.search(r"CUSTOMER\s+(.*)", line, re.IGNORECASE)
            if m:
                header["CUSTOMER"] = m.group(1).strip()
        if "RATING" in u:
            match = re.search(r"RATING\s*(.*?MVA.*?kV)", line, re.IGNORECASE)
            if match:
                header["RATING"] = match.group(1).strip()
    return header

# ================= HARDWARE PARSER =================
def parse_hardware(lines):
    data = []
    for line in lines:
        if re.search(r"(BOLT|STUD)", line.upper()) and re.search(r"M\d+X\d+", line.upper()):
            parts = line.split()
            for i, part in enumerate(parts):
                if re.match(r"S?-?M\d+X\d+-?[A-Z]*", part.upper()):
                    try:
                        data.append({
                            "FASTENER": parts[i - 1].upper(),
                            "SIZE_MATERIAL": part.upper(),
                            "QTY": int(parts[i + 1]),
                            "NUT": int(parts[i + 2]),
                            "L_NUT": int(parts[i + 3]),
                            "P_WASHER": int(parts[i + 4]),
                            "S_WASHER": int(parts[i + 5])
                        })
                    except:
                        continue
    return pd.DataFrame(data)

# ================= SUMMARY =================
def build_summary(df):
    summary = defaultdict(lambda: {
        "FULL_NUT": 0,
        "LOCK_NUT": 0,
        "PLAIN_WASHER": 0,
        "SPRING_WASHER": 0
    })

    unknown_materials = set()

    for _, row in df.iterrows():
        size_match = re.search(r"M\d+", row["SIZE_MATERIAL"].upper())
        if not size_match:
            continue

        size = size_match.group(0)
        material = "HDG"

        material_match = re.search(r"-([A-Z]+)$", row["SIZE_MATERIAL"].upper())
        if material_match:
            extracted = material_match.group(1)
            if extracted in ["HDG", "MAGNI"]:
                material = extracted
            else:
                unknown_materials.add(extracted)
        else:
            unknown_materials.add("UNKNOWN")

        key = (size, material)
        summary[key]["FULL_NUT"] += row["NUT"]
        summary[key]["LOCK_NUT"] += row["L_NUT"]
        summary[key]["PLAIN_WASHER"] += row["P_WASHER"]
        summary[key]["SPRING_WASHER"] += row["S_WASHER"]

    return summary, unknown_materials

# ================= SELECT FILE FUNCTION =================
def select_file():
    file_path = filedialog.askopenfilename(
        title="Select PDF",
        filetypes=[("PDF Files", "*.pdf")]
    )
    if not file_path:
        return

    lines = extract_pdf_lines(file_path)
    header = extract_header_info(lines)
    df = parse_hardware(lines)

    if df.empty:
        messagebox.showerror("Error", "No hardware data found.")
        return

    summary, unknown_materials = build_summary(df)

    if unknown_materials:
        messagebox.showwarning(
            "Material Warning",
            f"Unknown material detected:\n{', '.join(unknown_materials)}"
        )

    export_excel(header, df, summary, open_var.get())
    messagebox.showinfo("Success", "Hardware Output Generated Successfully!")

    try:
        winsound.PlaySound("startup.wav", winsound.SND_ASYNC)
    except:
        pass

    splash.after(2500, splash.destroy)
    splash.mainloop()
